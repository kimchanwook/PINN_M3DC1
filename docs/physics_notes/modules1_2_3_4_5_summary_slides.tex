\setmodule{Module 1}{1 of 6}
\begin{frame}[t]{Module 1: reduced resistive MHD model}
\bodyfs
The project uses a deliberately small parent model before making any M3D-C1 claim:
\[
(x,y,t)\in[0,2\pi)\times[0,2\pi)\times[0,T],
\qquad
\vect q=(\psi,U),
\qquad
\vect\mu=(\eta,\nu,A_0).
\]
\begin{itemize}
  \item Two-dimensional Cartesian domain: all resolved variation is in $(x,y)$; no $z$ dependence is solved.
  \item Constant density plus incompressibility: the flow is represented by one stream function $U$.
  \item Strong guide-field ordering: the total magnetic field is a large uniform $B_g\widehat{\vect z}$ plus a smaller in-plane perturbation $\vect B_\perp$.
  \item Version-1 trajectories are synthetic reduced-MHD data, not direct M3D-C1 output.
\end{itemize}
\[
\text{goal: }(x,y,t,\eta,\nu,A_0)\mapsto(\psi,U),
\qquad
\text{then derive }J,\omega,\vect B_\perp,\vect v_\perp.
\]
\end{frame}

\setmodule{Module 1}{2 of 6}
\begin{frame}[t]{Module 1: potentials, fields, and signs}
\bodyfs
Two scalar potentials encode the divergence-free physical fields:
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{Magnetic perturbation}
\[
\vect B_\perp=\nabla\psi\times\widehat{\vect z}
=(\psi_y,-\psi_x),
\quad
\nabla\cdot\vect B_\perp=0.
\]
\[
J=-\nabla^2\psi,
\qquad
\text{out-of-plane current density.}
\]
\end{column}
\begin{column}{0.49\textwidth}
\textbf{Center-of-mass flow}
\[
\vect v_\perp=\widehat{\vect z}\times\nabla U
=(-U_y,U_x),
\quad
\nabla\cdot\vect v_\perp=0.
\]
\[
\omega=\nabla^2U,
\qquad
\text{out-of-plane vorticity.}
\]
\end{column}
\end{columns}
\[
\vect B=B_g\widehat{\vect z}+\vect B_\perp,
\qquad
B_g=\text{constant background},
\qquad
\psi=\text{evolving perturbation flux}.
\]
Only $(\psi,U)$ are independent neural outputs.  Penalizing or reporting $J$ and $\omega$ tests derivative fidelity, not extra labels.
\end{frame}

\setmodule{Module 1}{3 of 6}
\begin{frame}[t]{Module 1: bracket form of advection}
\bodyfs
\[
\boxed{\bracket{f}{g}=f_xg_y-f_yg_x}
\]
The stream-function velocity turns material advection into a Poisson bracket:
\[
\vect v_\perp\cdot\nabla f=(-U_y)f_x+(U_x)f_y=\bracket{U}{f},
\qquad
\frac{Df}{Dt}=f_t+\bracket{U}{f}.
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{Algebraic properties}
\[
\bracket{f}{g}=-\bracket{g}{f},
\qquad
\bracket{f}{f}=0.
\]
\[
\int_\Omega h\bracket{f}{g}\dd A
=\int_\Omega f\bracket{g}{h}\dd A
\quad(\text{periodic}).
\]
\end{column}
\begin{column}{0.49\textwidth}
\textbf{Physical reading}
\begin{itemize}
  \item $U$ labels the pattern of fluid circulation.
  \item $\psi$ labels in-plane magnetic contours.
  \item Brackets move contours; diffusion changes them.
\end{itemize}
\end{column}
\end{columns}
Nonlinear brackets redistribute structure and exchange magnetic/kinetic energy; the explicit dissipators set irreversible loss.
\end{frame}

\setmodule{Module 1}{4 of 6}
\begin{frame}[t]{Module 1: governing equations and term meanings}
\bodyfs
\[
\boxed{\psi_t+\bracket{U}{\psi}=\eta\nabla^2\psi=-\eta J}
\]
\[
\boxed{\omega_t+\bracket{U}{\omega}=\bracket{J}{\psi}+\nu\nabla^2\omega},
\qquad
\omega=\nabla^2U,
\qquad
J=-\nabla^2\psi.
\]
\begin{itemize}
  \item $\bracket{U}{\psi}$: magnetic-flux contours are carried by the center-of-mass flow.
  \item $\eta\nabla^2\psi$: resistive diffusion lets plasma and field lines slip; this is where topology can change.
  \item $\bracket{U}{\omega}$: vorticity transport by the same incompressible velocity field.
  \item $\bracket{J}{\psi}$: reduced form of magnetic forcing from $\vect J\times\vect B$.
  \item $\nu\nabla^2\omega$: viscous smoothing of shear and vorticity.
\end{itemize}
The two equations are closed because $(J,\omega,\vect B_\perp,\vect v_\perp)$ are derivatives of $(\psi,U)$.
\end{frame}

\setmodule{Module 1}{5 of 6}
\begin{frame}[t]{Module 1: normalization and dimensionless groups}
\bodyfs
Use Alfv\'en scales before any machine-learning standardization:
\[
V_A=\frac{B_0}{\sqrt{\mu_0\rho_0}},
\qquad
 t_A=\frac{L_0}{V_A},
\qquad
x=\frac{x_d}{L_0},
\quad t=\frac{t_d}{t_A}.
\]
\[
\eta=\frac{\eta_m}{L_0V_A},
\qquad
\nu=\frac{\nu_d}{L_0V_A},
\qquad
S=\eta^{-1},
\quad \mathrm{Re}=\nu^{-1},
\quad \mathrm{Pm}=\frac{\nu}{\eta}.
\]
\begin{itemize}
  \item $S=t_\eta/t_A$ compares resistive diffusion to Alfv\'enic motion; large $S$ means slowly diffusing magnetic flux.
  \item $\mathrm{Re}=t_\nu/t_A$ compares viscous diffusion to flow advection; large $\mathrm{Re}$ permits sharp velocity gradients.
  \item $\mathrm{Pm}=\nu/\eta$ controls whether current sheets and vorticity sheets smooth on comparable scales.
\end{itemize}
\[
\text{physical nondimensionalization }\ne\text{ neural standardization }(q-m_q)/s_q.
\]
\end{frame}

\setmodule{Module 1}{6 of 6}
\begin{frame}[t]{Module 1: boundary, gauge, and energy checks}
\bodyfs
Periodic faces and all PDE-required derivatives match:
\[
q(0,y,t)=q(2\pi,y,t),
\quad
q(x,0,t)=q(x,2\pi,t),
\quad
q\in\{\psi,U,J,\omega\}.
\]
The stream-function gauge and magnetic mean condition are
\[
\boxed{\average{U}(t)=0},
\qquad
\boxed{\frac{\mathrm d}{\mathrm dt}\average{\psi}=0}.
\]
The continuous periodic energy law is
\[
\boxed{
\frac{\mathrm d}{\mathrm dt}\frac12\int_\Omega
\left(|\nabla\psi|^2+|\nabla U|^2\right)\dd A
=-\eta\int_\Omega J^2\dd A-\nu\int_\Omega\omega^2\dd A\le0.}
\]
A valid solver or PINN should not create energy from the bracket terms.  Energy defects isolate sign errors, derivative errors, and broken periodicity.
\end{frame}

\setmodule{Module 2}{1 of 6}
\begin{frame}[t]{Module 2: verified trajectory generation}
\bodyfs
For each parameter vector, Module 2 produces an accepted numerical realization
\[
\vect\mu=(\eta,\nu,A_0),
\qquad
\mathcal Q_{\vect\mu}^{N_x,N_y,\Delta t}
=\{\psi,U,J,\omega\}_{n i j}.
\]
The advanced Fourier state is only
\[
\widehat{\vect q}(t)=
\begin{bmatrix}\widehat\psi(t)\\ \widehat\omega(t)\end{bmatrix},
\qquad
\widehat U=-\frac{\widehat\omega}{k^2}\ (k\ne0),
\qquad
\widehat J=k^2\widehat\psi.
\]
\begin{itemize}
  \item Zero modes enforce $\widehat U_{00}=\widehat\omega_{00}=0$ and preserve the gauge/compatibility conditions.
  \item $(U,J)$ are reconstructed inside every RK stage, not merely after the step.
  \item A run becomes training data only after operator tests, trajectory checks, convergence evidence, and acceptance flags.
\end{itemize}
\end{frame}

\setmodule{Module 2}{2 of 6}
\begin{frame}[t]{Module 2: island-coalescence benchmark}
\bodyfs
The baseline periodic initial condition is
\[
\boxed{\psi_0=\sin x\sin y},
\qquad
\boxed{U_0=A_0(\cos x-\cos y)},
\qquad
J_0=2\psi_0,
\qquad
\omega_0=-U_0.
\]
\[\vect v_{\perp,0}=(-A_0\sin y,-A_0\sin x).\]
The flow drives like-signed current islands toward the central X-point.
\[
\begin{aligned}
\text{approach}&\rightarrow\text{compression}\rightarrow\text{current sheet}\\
&\rightarrow\text{resistive reconnection}\rightarrow\text{merged topology}.
\end{aligned}
\]
\begin{itemize}
  \item Baseline: $\eta=\nu=10^{-3}$, $A_0=0.1$, $N_x=N_y=128$, $\Delta t=10^{-3}$, $T\approx8$.
  \item The case exercises all terms: bracket transport, magnetic forcing, resistive diffusion, and viscous diffusion.
  \item The horizon must cover the first resolved peak and a post-peak interval, not just early smooth motion.
\end{itemize}
\end{frame}

\setmodule{Module 2}{3 of 6}
\begin{frame}[t]{Module 2: periodic Fourier operators}
\bodyfs
Use the half-open grid
\[
x_j=2\pi j/N_x,
\qquad
y_\ell=2\pi \ell/N_y,
\qquad
j=0,\ldots,N_x-1,
\quad \ell=0,\ldots,N_y-1.
\]
No endpoint is duplicated, so periodic differentiation is algebraic:
\[
\widehat{f_x}=ik_x\widehat f,
\qquad
\widehat{f_y}=ik_y\widehat f,
\qquad
\widehat{\nabla^2 f}=-k^2\widehat f.
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{Linear pieces}
\[
\widehat U=-\widehat\omega/k^2,
\qquad
\widehat J=k^2\widehat\psi.
\]
They are exact for resolved modes.
\end{column}
\begin{column}{0.49\textwidth}
\textbf{Nonlinear pieces}
\[
\bracket{f}{g}=f_xg_y-f_yg_x.
\]
Derivatives are spectral; products are evaluated on the grid.
\end{column}
\end{columns}
This Fourier--grid--Fourier loop defines the pseudo-spectral method.
\end{frame}

\setmodule{Module 2}{4 of 6}
\begin{frame}[t]{Module 2: dealiasing and time integration}
\bodyfs
Quadratic products can alias unresolved high modes into false low modes.  The frozen nonlinear operator is
\[
\boxed{\widehat{\mathcal B(f,g)}=\mathcal P_{2/3}\mathcal F(f_xg_y-f_yg_x)},
\qquad
|m|\le\lfloor N_x/3\rfloor,
\quad
|n|\le\lfloor N_y/3\rfloor.
\]
The semi-discrete dynamics are advanced by classical RK4:
\[
\vect q^{n+1}=\vect q^n+\frac{\Delta t}{6}
(\vect k_1+2\vect k_2+2\vect k_3+\vect k_4),
\qquad
\vect k_i=\mathcal L(\text{stage state}).
\]
At each stage:
\[
(\psi,\omega)\rightarrow(U,J)\rightarrow\text{spectral derivatives}\rightarrow\text{grid products}\rightarrow
\mathcal P_{2/3}\rightarrow\text{zero-mode enforcement}.
\]
Stability monitors are useful, but they do not replace explicit temporal refinement.
\end{frame}

\setmodule{Module 2}{5 of 6}
\begin{frame}[t]{Module 2: diagnostics for signs, topology, and resolution}
\bodyfs
Energy balance is monitored through the integrated defect
\[
\boxed{
\delta_E(t)=
\frac{\left|E(t)-E(0)+\int_0^t
\left[\eta\int J^2\dd A+\nu\int\omega^2\dd A\right]\dd s\right|}
{\max(E(0),\epsilon_{\rm mach})}.}
\]
Topology is summarized by O/X-point flux difference:
\[
\Delta\psi_{OX}=\psi_X-\psi_O,
\qquad
\Phi_{\rm rec}(t)=\Delta\psi_{OX}(0)-\Delta\psi_{OX}(t),
\qquad
\mathcal R_{\rm rec}=\frac{\dd\Phi_{\rm rec}}{\dd t}.
\]
Additional acceptance quantities include
\[
\|J\|_\infty,
\quad
\text{sheet width in grid cells},
\quad
\sum_{\text{tail}}|\widehat q_k|^2,
\quad
\average{U},\average{\psi},
\quad
\|J+\nabla^2\psi\|,
\quad
\|\omega-\nabla^2U\|.
\]
A visually plausible contour plot is not enough.
\end{frame}

\setmodule{Module 2}{6 of 6}
\begin{frame}[t]{Module 2: acceptance before data release}
\bodyfs
Verification proceeds from operators to full trajectories:
\[
\begin{aligned}
\text{operators}&\rightarrow\text{analytic fields}\rightarrow\text{bracket identities}\rightarrow\text{exact decay}\\
&\rightarrow\text{manufactured solution}\rightarrow\text{energy balance}\rightarrow\text{refinement}\\
&\rightarrow\text{Python/MATLAB parity}\rightarrow\text{case acceptance}.
\end{aligned}
\]
\begin{itemize}
  \item Temporal refinement should show fourth-order behavior until spatial error dominates.
  \item Spatial refinement should stabilize peak time, reconnection history, sheet width, spectra, and derived fields.
  \item Accepted files store fields, coordinates, parameters, diagnostics, config, revision, host metadata, and flags.
  \item Failed or marginal runs can be retained as diagnostics, but not silently mixed into the canonical training release.
\end{itemize}
\textbf{Data quality rule:} unverified simulation output is not a reference solution.
\end{frame}

\setmodule{Module 3}{1 of 5}
\begin{frame}[t]{Module 3: physical case versus numerical realization}
\bodyfs
Module 3 preserves identity so later model comparisons are auditable.
\begin{columns}[T,onlytextwidth]
\begin{column}{0.49\textwidth}
\textbf{Physical case}
\[
\code{physical\_case\_id}
\]
Equations, domain, initial/boundary conditions, physical parameters, and intended horizon.
\end{column}
\begin{column}{0.49\textwidth}
\textbf{Numerical realization}
\[
\code{case\_id}
\]
Solver, grid, time step, precision, language, configuration, code revision, and verification outcome.
\end{column}
\end{columns}
\[
\text{release}\supset\text{physical cases}\supset\text{realizations}\supset\text{snapshots}\supset\text{grid points}.
\]
\begin{itemize}
  \item Refined, restart, parity, or migrated realizations remain linked to the same physical case.
  \item Splitting by snapshots or grid points would leak a trajectory into both training and testing.
  \item The contract protects the distinction between ``M3D-C1-shaped'' and ``M3D-C1-derived.''
\end{itemize}
\end{frame}

\code{schema\_name=pinn\_m3dc1.case},
\qquad
\code{schema\_version=1.0.0},
\qquad

\begin{columns}[T,onlytextwidth]

\end{column}
\end{columns}
\[
q[n,i,j]=q(t_n,x_i,y_j),
\qquad
q\in\{\psi,U,J,\omega\},
\qquad
\text{reference arrays are finite float64 values.}
\]
A non-square, non-symmetric sentinel detects silent Python--MATLAB axis transposes.
\end{frame}

\setmodule{Module 3}{3 of 5}
\begin{frame}[t]{Module 3: whole-case splits and leakage control}
\bodyfs
Generalization is measured on disjoint physical-case sets:
\[
\mathcal P_{\rm all}=\mathcal P_{\rm tr}\cup\mathcal P_{\rm va}\cup\mathcal P_{\rm ti}\cup\mathcal P_{\rm te},
\qquad
\mathcal P_a\cap\mathcal P_b=\varnothing\quad(a\ne b).
\]
\begin{itemize}
  \item Training cases update weights and fit preprocessing; validation selects checkpoints and hyperparameters.
  \item Interpolation tests lie inside the training convex hull; extrapolation tests lie outside it.
  \item All realizations sharing one \code{physical\_case\_id} stay in one partition.
  \item The split manifest is frozen before model development and contains random seed, date, rule, and case lists.
\end{itemize}
\[
\widehat{\vect\alpha}=\operatorname{FitTransform}(\mathcal D_{\rm tr}),
\qquad
\text{then apply }\widehat{\vect\alpha}\text{ to validation/test data without refitting.}
\]
Refitting normalization or PCA-like transforms on held-out trajectories leaks field information.
\end{frame}

\setmodule{Module 3}{4 of 5}
\begin{frame}[t]{Module 3: learning views leave truth immutable}
\bodyfs
The canonical release stores physical arrays; learning views are reproducible derived products.  For
\[
(x,y,t,\eta,\nu,A_0)\mapsto(\psi,U),
\]
periodic coordinates are encoded as
\[
\vect e_{\rm per}=(\sin\phi_x,\cos\phi_x,\sin\phi_y,\cos\phi_y),
\qquad
\phi_x=2\pi x/L_x,
\quad
\phi_y=2\pi y/L_y.
\]
Positive transport coefficients can be transformed by
\[
r_\eta=\log_{10}\eta,
\qquad
r_\nu=\log_{10}\nu,
\qquad
\widetilde q=\frac{q-m_q}{s_q},
\quad q=m_q+s_q\widetilde q.
\]
PINN derivatives require recorded chain-rule factors:
\[
\partial_xq=\frac{s_q}{s_x}\partial_{\widetilde x}\widetilde q,
\qquad
\partial_x^2q=\frac{s_q}{s_x^2}\partial_{\widetilde x}^2\widetilde q,
\qquad
\partial_tq=\frac{s_q}{s_t}\partial_{\widetilde t}\widetilde q.
\]
\end{frame}

\setmodule{Module 3}{5 of 5}
\begin{frame}[t]{Module 3: protecting the M3D-C1 claim}
\bodyfs
Validation is layered:
\[
\text{structure}\rightarrow\text{relations}\rightarrow\text{numerics}\rightarrow\text{physics}\rightarrow\text{provenance}\rightarrow\text{release}.
\]
Readers check axes, finite values, gauges, derived-field identities, diagnostics, checksums, disjoint splits, parent links, and migration history.
\[
\boxed{
\begin{aligned}
\text{M3D-C1-shaped}={}&\text{metadata}+\text{mesh/coordinates}+\text{parameters}\\
&+\text{fields}+\text{diagnostics}+\text{splits}+\text{provenance}.
\end{aligned}}
\]
\begin{itemize}
  \item Unknown major schema versions are rejected; migrations create traceable new files.
  \item Genuine M3D-C1 exports remain source-native; adapters must document geometry, units, signs, gauges, projections, closures, and information loss.
  \item The project can claim interface readiness before it can claim physical equivalence to M3D-C1.
\end{itemize}
\end{frame}

\setmodule{Module 4}{1 of 4}
\begin{frame}[t]{Module 4: data-only field surrogate}
\bodyfs
The baseline neural surrogate approximates the parameterized field family
\[
\boxed{
\mathcal N_{\vect\theta}:
(x,y,t,\eta,\nu,A_0)\longmapsto(\widehat\psi,\widehat U).}
\]
Complete cases, not individual points, define the split:
\[
\vect\mu_c=(\eta_c,\nu_c,A_{0,c}),
\qquad
\mathcal D_c=\{(x_i,y_j,t_n,\vect\mu_c;\psi,U)\}_{n i j}.
\]
\begin{itemize}
  \item The model sees labels only through sampled field values of $(\psi,U)$.
  \item No PDE residual, derivative target, energy law, current/vorticity label, or boundary penalty appears in training.
  \item Time interpolation within $[0,T]$ is not forecasting beyond the simulated horizon.
  \item This baseline is the controlled comparator for asking whether physics information helps.
\end{itemize}
\end{frame}

\setmodule{Module 4}{2 of 4}
\begin{frame}[t]{Module 4: input representation and architecture}
\bodyfs
Periodic coordinates respect the domain before the network sees them:
\[
\phi_x=\frac{2\pi x}{L_x},
\qquad
\phi_y=\frac{2\pi y}{L_y},
\]
\[
\vect z=(\sin\phi_x,\cos\phi_x,\sin\phi_y,\cos\phi_y,
\widetilde t,\widetilde r_\eta,\widetilde r_\nu,\widetilde A_0).
\]
\begin{itemize}
  \item Smooth functions of $(\sin\phi,\cos\phi)$ are periodic in value and derivatives.
  \item $r_\eta=\log_{10}\eta$ and $r_\nu=\log_{10}\nu$ make decade-scale transport coefficients easier to learn.
  \item Scaling statistics are fit only on training cases.
  \item Baseline architecture: six fully connected $\tanh$ layers of width $128$ plus a linear two-output head.
\end{itemize}
Module 5 retains the same architecture family and representation so the comparison isolates the training objective.
\end{frame}

\setmodule{Module 4}{3 of 4}
\begin{frame}[t]{Module 4: labeled loss only}
\bodyfs
With standardized errors
\[
e_{\psi,n}=\widehat{\widetilde\psi}_n-\widetilde\psi_n,
\qquad
e_{U,n}=\widehat{\widetilde U}_n-\widetilde U_n,
\]
the case-balanced objective is
\[
\boxed{
\mathcal L_{\rm train}=\mathcal L_{\rm data}
=\frac{1}{|\mathcal C_{\mathcal B}|}
\sum_{c\in\mathcal C_{\mathcal B}}
\left[
\frac{1}{N_c}\sum_{n\in c}e_{\psi,n}^2
+\frac{1}{N_c}\sum_{n\in c}e_{U,n}^2
\right].}
\]
\begin{itemize}
  \item Case balancing prevents long trajectories or dense grids from dominating the objective.
  \item Adam, mini-batches, early stopping, and regularization are numerical choices, not physics information.
  \item Any good behavior in $J$, $\omega$, residuals, or energy must emerge indirectly from fitting $(\psi,U)$.
\end{itemize}
\end{frame}

\setmodule{Module 5}{1 of 5}
\begin{frame}[t]{Module 5: PDE residuals enter training}
\bodyfs
The network still predicts only $(\widehat\psi,\widehat U)$, but automatic differentiation constructs
\[
\widehat J=-\nabla^2\widehat\psi,
\qquad
\widehat\omega=\nabla^2\widehat U.
\]
The local reduced-MHD defects are
\[
\boxed{\Rpsi=\widehat\psi_t+\bracket{\widehat U}{\widehat\psi}-\eta\nabla^2\widehat\psi},
\]
\[
\boxed{\Romega=\widehat\omega_t+\bracket{\widehat U}{\widehat\omega}-\bracket{\widehat J}{\widehat\psi}-\nu\nabla^2\widehat\omega}.
\]
\begin{itemize}
  \item Labeled points constrain agreement with accepted trajectories.
  \item Collocation points constrain the equation even where no solver label is used.
  \item The residuals must be computed in physical coordinates after undoing scaling through the chain rule.
\end{itemize}
\end{frame}

\setmodule{Module 5}{2 of 5}
\begin{frame}[t]{Module 5: total objective and constraint roles}
\bodyfs
\[
\boxed{
\mathcal L_{\rm total}=w_{\rm data}\mathcal L_{\rm data}
+w_{\rm PDE}\mathcal L_{\rm PDE}
+w_{\rm IC}\mathcal L_{\rm IC}
+w_{\rm BC}\mathcal L_{\rm BC}
+w_{\rm global}\mathcal L_{\rm global}.}
\]
\begin{columns}[T,onlytextwidth]
\begin{column}{0.48\textwidth}
\textbf{Local terms}
\[
\mathcal L_{\rm PDE}=\average{\Rpsi^2/\sigma_\psi^2+\Romega^2/\sigma_\omega^2}_{\rm colloc}
\]
\[
\mathcal L_{\rm data}=\text{same labeled loss as Module 4.}
\]
\end{column}
\begin{column}{0.50\textwidth}
\textbf{Condition terms}
\begin{itemize}
  \item $\mathcal L_{\rm IC}$ selects the intended trajectory at $t=0$.
  \item $\mathcal L_{\rm BC}=0$ for hard periodic inputs; otherwise it measures face mismatch.
  \item $\mathcal L_{\rm global}$ contains gauge, mean-flux, and optional balance constraints.
\end{itemize}
\end{column}
\end{columns}
Weights are hyperparameters; they must be reported with normalization choices and collocation counts.
\end{frame}

\setmodule{Module 5}{3 of 5}
\begin{frame}[t]{Module 5: derivative order and gauge issues}
\bodyfs
Because $\widehat\omega=\nabla^2\widehat U$,
\[
\widehat\omega_t\sim\partial_t\partial_x^2\widehat U,
\qquad
\bracket{\widehat U}{\widehat\omega}\sim\partial_x^3\widehat U,
\qquad
\nabla^2\widehat\omega=\nabla^4\widehat U.
\]
The Lorentz bracket includes third derivatives of $\widehat\psi$ through $\widehat J=-\nabla^2\widehat\psi$.
\begin{itemize}
  \item Smooth activations such as $\tanh$ are required; ReLU-like nonsmoothness is incompatible with fourth derivatives.
  \item Scaling factors must remain in the differentiable graph; otherwise residual units and magnitudes are wrong.
  \item Automatic-differentiation derivatives should be tested against analytic periodic functions before training.
\end{itemize}
The PDE does not identify a spatially constant offset in $U$:
\[
\boxed{\average{\widehat U}(t,\vect\mu)=0.}
\]
\end{frame}

\setmodule{Module 5}{4 of 5}
\begin{frame}[t]{Module 5: global balances and curricula}
\bodyfs
A useful optional balance penalty is based on the exact energy law:
\[
\boxed{
\frac{\mathrm d}{\mathrm dt}\frac12\int_\Omega
\left(|\nabla\widehat\psi|^2+|\nabla\widehat U|^2\right)\dd A
+\eta\int_\Omega\widehat J^2\dd A
+\nu\int_\Omega\widehat\omega^2\dd A=0.}
\]
Collocation defines what ``PDE accuracy'' means:
\[\mathcal X_{\rm colloc}=\{(x,y,t,\eta,\nu,A_0)\}.\]
Fixed validation collocation and adaptive training collocation are different objects.
Practical curriculum:
\[
\begin{aligned}
\text{derivative tests}&\rightarrow\text{data warm start}\rightarrow\Rpsi\rightarrow\Romega\\
&\rightarrow\text{longer windows}\rightarrow\text{multi-parameter cases}.
\end{aligned}
\]
Global penalties should follow quadrature verification; otherwise they can enforce a numerically wrong balance.
\end{frame}

\setmodule{Module 5}{5 of 5}
\begin{frame}[t]{Module 5: scientific interpretation of the PINN result}
\bodyfs
Compare matched models on unseen cases using
\[
\begin{gathered}
\varepsilon_{L_2}^{(\psi,U,J,\omega)},\quad \Rpsi,\Romega,\quad \delta_E,\\
\Phi_{\rm rec},\mathcal R_{\rm rec},\quad \text{training and inference cost}.
\end{gathered}
\]
A PINN can help by improving labeled-data efficiency, derivative fidelity, parameter generalization, or physical admissibility.

It can also fail through slow training, gradient conflict, oversmoothed sheets, residual imbalance, or wrong/incomplete physics.

\textcolor{WarningRed}{\textbf{Conclusion standard:}} the PINN is not assumed to win.\par
The result remains a surrogate of synthetic two-field reduced-MHD dynamics.\par
Full-code surrogate claim: genuine M3D-C1 data plus a validated adapter.
\end{frame}
